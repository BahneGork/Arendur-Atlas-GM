# Items not Identified, (tasks)
```dataview 
TABLE WITHOUT ID (filter(file.tasks, (t) => t.completed = "incompleted").text) AS Task, choice(Tasks.completed, "Identified", "Not identified") AS Status, file.link AS "File" 
FROM "01 GM/Campaigns/The Arendur Rising Campaign/The Shadow Company/Loot/Import items"
WHERE file.tasks
FLATTEN file.tasks AS Tasks
sort name asc
```
